// a. Find all cars sold by brand Toyota
db.cars.find({ brand: "Toyota" })

// b. Find all cars with a price greater than 30000
db.cars.find({ price: { $gt: 30000 } })

// c. Return the car_id, brand, and price of all cars sorted by price descending
db.cars.find(
  {},
  { car_id: 1, brand: 1, price: 1, _id: 0 }
).sort({ price: -1 })

// d. Find all Ford cars sold after 2024-07-01
db.cars.find({
  brand: "Ford",
  sold_date: { $gt: ISODate("2024-07-01") }
})

// e. Count how many cars were sold by each brand
db.cars.aggregate([
  {
    $group: {
      _id: "$brand",
      totalSold: { $sum: 1 }
    }
  }
])

// f. Find the total revenue from car sales
db.cars.aggregate([
  {
    $group: {
      _id: null,
      totalRevenue: { $sum: "$price" }
    }
  }
])

// g. Project only the brand, model, and sold_date for all documents
db.cars.find(
  {},
  { brand: 1, model: 1, sold_date: 1, _id: 0 }
)

// h. Use $match and $group to find the total price of Toyota cars sold
db.cars.aggregate([
  {
    $match: { brand: "Toyota" }
  },
  {
    $group: {
      _id: "$brand",
      totalPrice: { $sum: "$price" }
    }
  }
])
